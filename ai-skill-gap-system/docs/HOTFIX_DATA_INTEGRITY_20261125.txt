SQL TỰ KIỂM TRA (CHỈ ĐỌC)
SELECT p.industry_id,COUNT(*) n FROM jobs j JOIN positions p
ON p.position_id=j.position_id GROUP BY p.industry_id;

SELECT r.batch_id,COUNT(DISTINCT j.job_id) n FROM jobs j
JOIN jobs_raw r ON r.job_id=j.raw_job_id GROUP BY r.batch_id;

SELECT j.job_id,r.batch_id FROM jobs j JOIN jobs_raw r ON r.job_id=j.raw_job_id
WHERE j.job_id<=1000 AND r.batch_id<>'BATCH_2026_Q2_VERIFIED';

SELECT p.industry_id,COUNT(DISTINCT j.job_id) n FROM jobs j
JOIN jobs_raw r ON r.job_id=j.raw_job_id JOIN positions p ON p.position_id=j.position_id
WHERE j.status='active' AND r.batch_id='BATCH_2026_Q2_VERIFIED'
GROUP BY p.industry_id;

SELECT c.batch_id,c.category,COUNT(DISTINCT w.worker_id) n FROM workers w
JOIN cvs_raw c ON c.cv_id=w.cv_raw_id WHERE w.is_active=1
GROUP BY c.batch_id,c.category;
